{{- define "RenderImage" -}} {{- /* Renders an image as a (WebP source plus original fallback) or as a plain . Params (dict): imgObj Required. Hugo image resource, or "" if not found src Required. Fallback src used by the plain alt Required. Alt text processable Required. Result of reflect.IsImageResourceProcessable optimize Optional. Truthy to enable the render loading Required. "eager" or "lazy" prio Optional. "high", "low" or "auto", omitted when empty w Optional. Width attribute, falls back to the intrinsic size h Optional. Height attribute, falls back to the intrinsic size Returns: None. Renders HTML directly. Note: The render is used only when optimize, imgObj and processable are all truthy. Otherwise the plain render is used. In the render, src is the original image (high resolution, with EXIF data, and available in PhotoSwipe) and srcset is the WebP version at the same resolution for tablets and desktops. The container is never full-width on desktop but is full-width on mobile, so the same source serves both. */ -}} {{- $imgObj := .imgObj -}} {{- $processable := .processable -}} {{- $loading := .loading -}} {{- $prio := .prio -}} {{- $width := .w -}} {{- $height := .h -}} {{- $noSize := not (or $width $height) -}} {{- if and .optimize $imgObj $processable -}} {{- $opt := partial "lib/img-opt.html" (dict "imgObj" $imgObj "desktopWidth" 1320 "mobileWidth" 1320) -}} {{- /* No w/h given: fall back to the size of the WebP version */ -}} {{- if $noSize -}} {{- $width = $opt.imgMobile.Width -}} {{- $height = $opt.imgMobile.Height -}} {{- end -}} {{ with $opt.imgMobile.RelPermalink }}{{ end }} {{- else -}} {{- /* No w/h given: fall back to the intrinsic size of the processable resource */ -}} {{- if and $noSize $processable -}} {{- $width = $imgObj.Width -}} {{- $height = $imgObj.Height -}} {{- end -}} {{- end -}} {{- end -}} {{- $resource := "" -}} {{- $src := .Destination | safeURL -}} {{- $u := urls.Parse .Destination -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $resource = or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- with $resource -}} {{- $src = .RelPermalink -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- $processable := reflect.IsImageResourceProcessable $resource -}} {{- /* Attributes consumed by img, not passed to figure */ -}} {{- $imgOnlyAttrs := slice "w" "h" "prio" "width" "height" "fetchpriority" "loading" -}} {{- /* Short names take priority, full names act as aliases */ -}} {{- $w := or (index .Attributes "w") (index .Attributes "width") -}} {{- $h := or (index .Attributes "h") (index .Attributes "height") -}} {{- /* prio only accepts valid values, anything else is ignored */ -}} {{- $prio := "" -}} {{- with or (index .Attributes "prio") (index .Attributes "fetchpriority") -}} {{- $v := lower (printf "%v" .) -}} {{- if in (slice "high" "low" "auto") $v -}} {{- $prio = $v -}} {{- else -}} {{- warnf "Unsupported fetchpriority setting: %s. Must be in one of \"high\", \"low\" or \"auto\"" $v }} {{- end -}} {{- else -}} {{- /* no default value, use browser's default priority */ -}} {{- end -}} {{- /* high defaults to eager, everything else defaults to lazy, loading can override manually */ -}} {{- $loading := cond (eq $prio "high") "eager" "lazy" -}} {{- with index .Attributes "loading" -}} {{- $v := lower (printf "%v" .) -}} {{- if in (slice "eager" "lazy") $v -}} {{- $loading = $v -}} {{- end -}} {{- end -}} {{- /**/ -}}
