{{- /* Computes the base path and content hash for an image, used to build cache-busted output filenames. Params (dict): imgObj Required. Hugo image resource Returns (dict): base path without extension hash 6-char xxhash of the image Note: Use w/h as the hash input because the .Content string can be millions of chars. */ -}} {{- $imgObj := . -}} {{- $decodedPath := urls.PathUnescape $imgObj.RelPermalink -}} {{- $base := path.Join (path.Dir $decodedPath) (path.BaseName $decodedPath) -}} {{- $hashSrc := printf "%s%d%d" $decodedPath $imgObj.Width $imgObj.Height -}} {{- $hash := substr (hash.XxHash $hashSrc) 0 6 -}} {{- return (dict "base" $base "hash" $hash) -}}
