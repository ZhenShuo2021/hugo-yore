{{- /* Inline SVG icon from assets/yore/icons. Accepts either a plain string (the icon name, backward compatible) or a dict. Params (dict): name (string, required) - icon name, with or without ".svg" label (string, optional) - accessible description. class (string, optional) - extra classes rotate (number, optional) - rotation in degrees, e.g. 90 flip (string, optional) - "x", "y" or "both" animation (string, optional) - "spin", "pulse" or "bounce". */ -}} {{- $p := dict -}} {{- if reflect.IsMap . -}} {{- $p = . -}} {{- else -}} {{- $p = dict "name" . -}} {{- end -}} {{- $flips := dict "x" "-scale-x-100" "y" "-scale-y-100" "both" "-scale-100" -}} {{- $animations := dict "spin" "motion-safe:animate-spin" "pulse" "motion-safe:animate-pulse" "bounce" "motion-safe:animate-bounce" -}} {{- $name := strings.TrimSuffix ".svg" ($p.name | default "") -}} {{- $icon := resources.GetMatch (printf "yore/icons/{**/,}%s.svg" $name) -}} {{- with $icon -}} {{- $cls := "yr-icon inline-flex" -}} {{- with $p.flip -}} {{- $c := index $flips . -}} {{- if not $c -}} {{- errorf "icon %q: unknown flip %q" $name . -}} {{- end -}} {{- $cls = printf "%s %s" $cls $c -}} {{- end -}} {{- with $p.animation -}} {{- $c := index $animations . -}} {{- if not $c -}} {{- errorf "icon %q: unknown animation %q" $name . -}} {{- end -}} {{- $cls = printf "%s %s" $cls $c -}} {{- end -}} {{- with $p.class -}} {{- $cls = printf "%s %s" $cls . -}} {{- end -}} {{- $style := "" -}} {{- with $p.rotate -}} {{- $style = printf "rotate: %ddeg" (int .) -}} {{- end -}} {{- .Content | safeHTML -}} {{- else -}} {{- warnf "icon %q not found" $name -}} {{- end -}}
